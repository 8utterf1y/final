3.3 L3 闭包指纹验证（防"脏缓存"）
def _file_fingerprint(db, file_ids):
    """对闭包内所有文件 (rel_path + sha256) 做 SHA-256"""
    h = hashlib.sha256()
    for fid in sorted(set(file_ids)):
        f = db.get_file(fid)
        h.update(f.rel_path.encode())
        h.update(f.sha256.encode())
    return h.hexdigest()
# 命中验证：
actual_fp = _file_fingerprint(db, 
    [obj.target.file_id] 
    + [s.file_id for s,_ in obj.direct_callees]
    + [s.file_id for s,_ in obj.transitive_callees]
    + [s.file_id for s,_ in obj.direct_callers]
    + [s.file_id for s,_ in obj.transitive_callers]
    + [s.file_id for s,_ in obj.symbol_dependencies])
if meta_path.read_text() == actual_fp:  # 指纹一致 → 有效缓存
    return obj
return None  # 任一被引用文件变了 → 重建

五、双向 BFS 调用链 + caller_depth 控制
5.1 BFS 实现（context/extractor.py）
def _bfs_layered(db, start_id, depth, edge_fn, visited):
    """edge_fn = db.callees_of() 或 db.callers_of()"""
    direct, transitive, edges = [], [], []
    layer = [start_id]
    for level in range(1, depth + 1):
        next_layer = []
        for parent in layer:
            for nb in edge_fn(parent):
                if nb.id in visited:
                    continue                # 防环
                visited.add(nb.id)
                edges.append((parent, nb.id, level))
                body = db.read_symbol_body(nb.id) if nb.kind == "function" else ""
                pair = (nb, body)
                (direct if level == 1 else transitive).append(pair)
                if nb.kind == "function":   # 只有函数能继续展开
                    next_layer.append(nb.id)
        layer = next_layer
        if not layer:
            break                            # 死路提前停止
    return direct, transitive, edges
关键设计：
- 只有 function 类型节点继续 BFS 展开（struct/macro 是叶子）
- visited 集合在 callee/caller 两个方向共享，防止重复包含
- 边记录 (parent, child, level) 用于 Mermaid 调用图渲染

def _merge_extra_body(base, chat_template_kwargs, enable_thinking):
    # 优先级（低→高）：base < chat_template_kwargs < enable_thinking
    out = copy.deepcopy(base) or {}
    if chat_template_kwargs:
        ctk = out.setdefault("chat_template_kwargs", {})
        ctk.update(chat_template_kwargs)
    if enable_thinking is not None:
        ctk = out.setdefault("chat_template_kwargs", {})
        ctk["enable_thinking"] = bool(enable_thinking)
    return out

def compute_cache_key(body, ctx_fingerprint, model_id, prompt_version,
                      rule_version, level, rule_scope):
    body_norm = normalize_body(body)   # \s+ → 单空格
    parts = [body_norm, ctx_fingerprint, model_id, 
             prompt_version, rule_version, level, rule_scope]
    return sha256("\0".join(parts)).hexdigest()
